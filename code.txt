from django.db import models
from django.contrib.auth.models import User
from shortuuid.django_fields import ShortUUIDField


class UserProfile(models.Model):
    user = models.OneToOneField(
        User,
        on_delete=models.CASCADE,
        related_name='profile',
        primary_key=True,
    )
    real_name = models.CharField(max_length=20, blank=True)
    phone = models.CharField(unique=True, max_length=11)
    role = models.CharField(max_length=32, default='viewer')
    id_card = models.CharField(unique=True, max_length=18, blank=True, null=True)
    gender = models.CharField(max_length=2, blank=True, null=True)
    status = models.CharField(max_length=32, default='正常')
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        db_table = 'user_profile'   # 你自己定表名

    def __str__(self):
        return self.user.username
